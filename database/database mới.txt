# QUẢN LÍ NHÂN SỰ
-- ===============================
-- BỘ PHẬN
-- ===============================
CREATE TABLE bophan (
    mabophan INT AUTO_INCREMENT PRIMARY KEY,
    tenbophan VARCHAR(200) NOT NULL
);

-- ===============================
-- NHÂN VIÊN
-- ===============================
CREATE TABLE nhanvien (
    manhanvien VARCHAR(20) PRIMARY KEY,
    tennhanvien VARCHAR(100) NOT NULL,
    ngaysinh DATE,
    sodienthoai VARCHAR(11),
    gioitinh ENUM('Nam','Nữ') NOT NULL,
    quequan VARCHAR(100) NOT NULL,
    mabophan INT NOT NULL,
    trangthai ENUM ('Đang làm việc', 'Đã nghỉ việc') NOT NULL DEFAULT 'Đang làm việc',
    FOREIGN KEY (mabophan) REFERENCES bophan(mabophan)
);

-- ===============================
-- TÀI KHOẢN
-- ===============================
CREATE TABLE taikhoan(
    mataikhoan INT AUTO_INCREMENT PRIMARY KEY,
    tentaikhoan VARCHAR(20) NOT NULL UNIQUE,
    matkhau VARCHAR(255) NOT NULL,
    manhanvien VARCHAR(20) UNIQUE,
    chucvu ENUM ('Admin','Bán hàng','Kho') NOT NULL,
    trangthai ENUM ('Đang hoạt động', 'Ngừng hoạt động') NOT NULL,
    CONSTRAINT chkadminnv
    CHECK (
        (chucvu = 'Admin' AND manhanvien IS NULL)
        OR
        (chucvu IN ('Bán hàng','Kho') AND manhanvien IS NOT NULL)
    ),
    FOREIGN KEY (manhanvien) REFERENCES nhanvien(manhanvien)
);


# QUẢN LÍ KHO

-- ===============================
-- DANH MỤC
-- ===============================
CREATE TABLE danhmuc (
    madanhmuc INT AUTO_INCREMENT PRIMARY KEY,
    tendanhmuc VARCHAR(100) NOT NULL,
    ghichu TEXT
);

-- ===============================
-- MẶT HÀNG
-- ===============================
CREATE TABLE mathang (
    mamathang INT AUTO_INCREMENT PRIMARY KEY,
    tenmathang VARCHAR(100) NOT NULL,
    soluong INT NOT NULL DEFAULT 0,
    dongiaban INT NOT NULL,
    madanhmuc INT NOT NULL,
    ghichu TEXT,
    FOREIGN KEY (madanhmuc) REFERENCES danhmuc(madanhmuc)
);

-- ===============================
-- NHÀ CUNG CẤP
-- ===============================
CREATE TABLE nhacungcap (
    manhacungcap INT AUTO_INCREMENT PRIMARY KEY,
    tennhacungcap VARCHAR(100) NOT NULL,
    masothue VARCHAR(100),
    sodienthoai VARCHAR(11) NOT NULL,
    diachi VARCHAR(100),
    ghichu TEXT
);

-- ===============================
-- ĐƠN NHẬP HÀNG
-- ===============================
CREATE TABLE donnhaphang (
    madonnhaphang INT AUTO_INCREMENT PRIMARY KEY,
    manhacungcap INT NOT NULL,
    ngaynhaphang DATE NOT NULL,
    mataikhoan INT NOT NULL,
    ghichu TEXT,
    FOREIGN KEY (mataikhoan) REFERENCES taikhoan(mataikhoan),
    FOREIGN KEY (manhacungcap) REFERENCES nhacungcap(manhacungcap)
);

-- ===============================
-- CHI TIẾT ĐƠN NHẬP HÀNG
-- ===============================
CREATE TABLE chitietdonnhaphang (
    machitietdonnhaphang INT AUTO_INCREMENT PRIMARY KEY,
    mamathang INT NOT NULL,
    madonnhaphang INT NOT NULL,
    soluong INT NOT NULL,
    dongianhap INT NOT NULL,
    FOREIGN KEY (madonnhaphang) REFERENCES donnhaphang(madonnhaphang) ON DELETE CASCADE,
    FOREIGN KEY (mamathang) REFERENCES mathang(mamathang)
);


# QUẢN LÍ BÁN HÀNG

-- ===============================
-- HÓA ĐƠN
-- ===============================
CREATE TABLE hoadon (
    mahoadon INT AUTO_INCREMENT PRIMARY KEY,
    ngayxuat DATE NOT NULL,
    mataikhoan INT NOT NULL,
    FOREIGN KEY (mataikhoan) REFERENCES taikhoan(mataikhoan)
);

-- ===============================
-- CHI TIẾT HÓA ĐƠN
-- ===============================
CREATE TABLE chitiethoadon (
    machitiethoadon INT AUTO_INCREMENT PRIMARY KEY,
    mahoadon INT NOT NULL,
    mamathang INT NOT NULL,
    soluong INT NOT NULL,
    dongiaban INT NOT NULL,
    FOREIGN KEY (mahoadon) REFERENCES hoadon(mahoadon) ON DELETE CASCADE,
    FOREIGN KEY (mamathang) REFERENCES mathang(mamathang)
);

-- ===============================
-- TRIGGERS
-- ===============================
DELIMITER //

-- Thêm chi tiết hóa đơn
CREATE TRIGGER themctdh
AFTER INSERT ON chitiethoadon
FOR EACH ROW
BEGIN
    UPDATE mathang
    SET soluong = soluong - NEW.soluong
    WHERE mamathang = NEW.mamathang;
END//

-- Xóa chi tiết hóa đơn
CREATE TRIGGER xoactdh
AFTER DELETE ON chitiethoadon
FOR EACH ROW
BEGIN
    UPDATE mathang
    SET soluong = soluong + OLD.soluong
    WHERE mamathang = OLD.mamathang;
END//

-- Thêm chi tiết đơn nhập hàng: cộng tồn kho
CREATE TRIGGER themctdnh
AFTER INSERT ON chitietdonnhaphang
FOR EACH ROW
BEGIN
    UPDATE mathang
    SET soluong = soluong + NEW.soluong
    WHERE mamathang = NEW.mamathang;
END//

-- Xóa chi tiết đơn nhập hàng: trừ tồn kho
CREATE TRIGGER xoactdnh
AFTER DELETE ON chitietdonnhaphang
FOR EACH ROW
BEGIN
    UPDATE mathang
    SET soluong = soluong - OLD.soluong
    WHERE mamathang = OLD.mamathang;
END//

DELIMITER ;
